88 auto it = m_keys.find(tt);
89 if (it == m_keys.end()) {

97 bool comp_equal(const Ordinal& a, const Ordinal& b) const {
98 return (!m_order(a, b) && !m_order(b, a));
99 }
100
101 bool eligible(const Ordinal& ord) const {
102 return m_order(ord, m_current) || comp_equal(ord, m_current);
103 }
104
105 bool check_key_impl(const key_type& key, Ordinal ord, ttg::TTBase *tt) {
106 if (!m_stopped) {
107 if (eligible(ord)) {
108 // key should be executed
109 if (m_auto_release) { // only needed for auto-release
110 m_active.fetch_add(1, std::memory_order_relaxed);
111 // revert the current ordinal to the lower ordinal
112 m_current = ord;
113 }
114 return true;
115 } else if (m_sequence.empty() && m_auto_release && 0 == m_active.load(std::memory_order_relaxed)) {
116 // there are no keys (active or blocked) so we execute to avoid a deadlock
117 // we don't change the current ordinal because there may be lower ordinals coming in later
118 // NOTE: there is a race condition between the check here and the increment above.
119 // This is mostly benign as it can lead to out-of-sequence released tasks.
120 // Avoiding this race would incur significant overheads.
121 m_active.fetch_add(1, std::memory_order_relaxed);
122 return true;
123 }
124 }
125 // key should be deferred
126 auto g = this->lock_guard();
127 if (!m_stopped && eligible(ord)) {
128 // someone released this ordinal while we took the lock
129 return true;
130 }
131 auto it = m_sequence.find(ord);
132 if (it == m_sequence.end()) {
133 auto [iter, success] = m_sequence.insert(std::make_pair(ord, sequence_elem_t{}));
134 assert(success);
135 it = iter;
136 }
137 it->second.add_key(key, tt);
138 return false;
139 }
140
141
142 void complete_key_impl() {
143 if (m_auto_release) {
144 auto active = m_active.fetch_sub(1, std::memory_order_relaxed) - 1;
145 if (0 == active) {
146 release_next();
147 }
148 }
149 }
150
151 // used in the auto case
152 void release_next() {
153 if (this->m_stopped) {
154 // don't release tasks if we're stopped
155 return;
156 }
157 // trigger the next sequence
158 sequence_elem_t elem;
159 {
160 // extract the next sequence
161 auto g = this->lock_guard();
162 auto it = this->m_sequence.begin(); // sequence is ordered by ordinal
163 if (it == this->m_sequence.end()) {
164 return; // nothing to be done
165 }
166 this->m_current = it->first;
167 elem = std::move(it->second);
168 this->m_sequence.erase(it);
169 }
170
171 for (auto& seq : elem.m_keys) {
172 // account for the newly active keys
173 this->m_active.fetch_add(seq.second.size(), std::memory_order_relaxed);
174 this->notify_listener(ttg::span<key_type>(seq.second.data(), seq.second.size()), seq.first);
175 }
176 }
177
178
179 // used in the non-auto case
180 void release_next(ordinal_type ord, bool force_check = false) {
181 if (this->m_stopped) {
182 // don't release tasks if we're stopped but remember that this was released
183 this->m_current = ord;
184 return;
185 }
186 if (!force_check && eligible(ord)) {
187 return; // already at the provided ordinal, nothing to be done
188 }
189 // trigger the next sequence(s) (m_sequence is ordered by ordinal)
190 std::vector<sequence_elem_t> seqs;
191 {
192 auto g = this->lock_guard();
193 // set current ordinal
194 this->m_current = ord;
195 for (auto it = this->m_sequence.begin(); it != this->m_sequence.end();) {
196 if (!eligible(it->first)) break;
197 // extract the next sequence
198 this->m_current = it->first;
199 seqs.push_back(std::move(it->second));
200 it = this->m_sequence.erase(it);
201 }
202 }
203 for (auto& elem : seqs) {
204 for (auto& e : elem.m_keys) {
205 // account for the newly active keys
206 this->notify_listener(ttg::span<key_type>(e.second.data(), e.second.size()), e.first);
207 }
208 }
209 }

305 void start() {
306 if (m_stopped) {
307 m_stopped = false;
308 if (m_auto_release) {
309 release_next();
310 } else {
311 auto ord = m_current;
312 // release the first set of available keys if none were set explicitly
313 if (ord == std::numeric_limits<ordinal_type>::min() &&
314 this->m_sequence.begin() != this->m_sequence.end()) {
315 ord = this->m_sequence.begin()->first;
316 }
317 release_next(ord, true); // force the check for a next release even if the current ordinal hasn't changed
318 }
319 }
320 }

326 if (m_auto_release) {
327 // last key for this ordinal, release the next
328 // the provided ordinal is ignored
329 release_next();
330 } else {

332 }
