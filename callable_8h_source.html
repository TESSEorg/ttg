42 // member function pointers: the (reference to the) object is the first argument, args_sans_object omits it.
43 // The macros below expand to one partial specialization per combination of the PMF's qualifiers:
44 // 4 cv (none, const, volatile, const volatile) x 3 ref (none, &, &&) x 2 noexcept x 2 C-style varargs = 48.
45 // OBJREF is the reference type of the object argument: & for unqualified and &-qualified PMFs, && for
46 // &&-qualified ones (e.g. `void (C::*)(int) const` -> args = typelist<const C&, int>).
47 // Abominable function types (e.g. `void(int) const`) are not callables and are deliberately not handled.
48#define TTG_CALLABLE_TRAITS_PMF_IMPL(CV, REF, OBJREF, NOEXCEPT, VARARGS) \
49 template <typename R, typename C, typename... Args> \
50 struct callable_traits_impl<R (C::*)(Args... VARARGS) CV REF NOEXCEPT> \
51 : callable_signature<R, CV C OBJREF, Args...> { \
52 using args_sans_object = ttg::typelist<Args...>; \
53 };
54#define TTG_CALLABLE_TRAITS_COMMA_ELLIPSIS , ...
55#define TTG_CALLABLE_TRAITS_PMF_VARARGS(CV, REF, OBJREF, NOEXCEPT) \
56 TTG_CALLABLE_TRAITS_PMF_IMPL(CV, REF, OBJREF, NOEXCEPT, ) \
57 TTG_CALLABLE_TRAITS_PMF_IMPL(CV, REF, OBJREF, NOEXCEPT, TTG_CALLABLE_TRAITS_COMMA_ELLIPSIS)
58#define TTG_CALLABLE_TRAITS_PMF_NOEXCEPT(CV, REF, OBJREF) \
59 TTG_CALLABLE_TRAITS_PMF_VARARGS(CV, REF, OBJREF, ) \
60 TTG_CALLABLE_TRAITS_PMF_VARARGS(CV, REF, OBJREF, noexcept)
61#define TTG_CALLABLE_TRAITS_PMF_REF(CV) \
62 TTG_CALLABLE_TRAITS_PMF_NOEXCEPT(CV, , &) \
63 TTG_CALLABLE_TRAITS_PMF_NOEXCEPT(CV, &, &) \
64 TTG_CALLABLE_TRAITS_PMF_NOEXCEPT(CV, &&, &&)

182 template <std::size_t N>
183 constexpr auto ordinal2index(std::size_t ordinal, std::array<std::size_t, N> extents) {
184 std::array<std::size_t, N> idx = {};
185 for (size_t d = 0; d < N; ++d) {
186 idx[d] = ordinal % extents[d];
187 ordinal /= extents[d];
188 }
189 return idx;
190 }

205 template <std::size_t Ordinal, typename Func, typename... Typelists, std::size_t... ArgIdx>
206 auto compute_arg_binding_types_impl(Func& func, typelist<Typelists...> argument_type_lists,
207 std::index_sequence<ArgIdx...> arg_idx = {}) {
208 using arg_typelists_t = typelist<Typelists...>;
209 constexpr auto Order = sizeof...(Typelists);
210 constexpr std::array<std::size_t, Order> extents = {
211 std::tuple_size_v<std::tuple_element_t<ArgIdx, arg_typelists_t>>...};
212 constexpr auto tensor_size = (extents[ArgIdx] * ...);
213 static_assert(tensor_size >= Ordinal);
214 if constexpr (tensor_size == Ordinal) {
215 return typelist<typelist<>, typelist<>>{};
216 } else {
217 constexpr auto idx = ordinal2index(Ordinal, extents);
218 auto args = typelist<std::tuple_element_t<idx[ArgIdx], std::tuple_element_t<ArgIdx, arg_typelists_t>>...>{};
219 using args_sans_void_t = drop_void_t<decltype(args)>;
220 if constexpr (is_invocable_typelist_v<Func, args_sans_void_t>) {
221 using return_type = invoke_result_typelist_t<Func, args_sans_void_t>;
222 return ttg::typelist<ttg::typelist<return_type>, decltype(args)>{};
223 } else {
224 return compute_arg_binding_types_impl<Ordinal + 1>(func, argument_type_lists, arg_idx);
225 }
226 }
227 }

243 template <std::size_t Ordinal, typename ReturnType, typename Func, typename... Typelists, std::size_t... ArgIdx>
244 auto compute_arg_binding_types_r_impl(Func& func, typelist<Typelists...> argument_type_lists,
245 std::index_sequence<ArgIdx...> arg_idx = {}) {
246 using arg_typelists_t = typelist<Typelists...>;
247 constexpr auto Order = sizeof...(Typelists);
248 constexpr std::array<std::size_t, Order> extents = {
249 std::tuple_size_v<std::tuple_element_t<ArgIdx, arg_typelists_t>>...};
250 constexpr auto tensor_size = (extents[ArgIdx] * ...);
251 static_assert(tensor_size >= Ordinal);
252 if constexpr (tensor_size == Ordinal) {
253 return typelist<>{};
254 } else {
255 constexpr auto idx = ordinal2index(Ordinal, extents);
256 auto args = typelist<std::tuple_element_t<idx[ArgIdx], std::tuple_element_t<ArgIdx, arg_typelists_t>>...>{};
257 if constexpr (is_invocable_typelist_r_v<ReturnType, Func, drop_void_t<decltype(args)>>) {
258 return args;
259 } else {
260 return compute_arg_binding_types_r_impl<Ordinal + 1, ReturnType>(func, argument_type_lists, arg_idx);
261 }

310 template <typename T, typename = void>
311 struct candidate_argument_bindings;

313 template <typename T>
314 struct candidate_argument_bindings<T, std::enable_if_t<!std::is_reference_v<T> && !std::is_void_v<T>>> {
315 using type = std::conditional_t<std::is_const_v<T>, typelist<const T&>,
316 typelist<
317 // RATIONALE for this order of binding detection tries:
318 // - to be able to distinguish arguments declared as auto& vs auto&& should try
319 // binding to T&& first since auto& won't bind to it
320 // - HOWEVER argument declared as const T& will bind to either T&& or const T&,
321 // so this order will detect such argument as binding to T&&, which will
322 // indicate to the runtime that the argument is CONSUMABLE and may cause
323 // creation of extra copies. Thus you should not try to use nongeneric
324 // data arguments in generic task functions; for purely nongeneric functions
325 // a different introspection mechanism (callable_traits) is used
326 T&&, const T&
327 // - no need to check T& since auto& and auto&& both bind to it
328 //, T&
329 >>;
330 };
