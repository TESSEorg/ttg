122 template<std::size_t i = 0, typename TT>
123 void init_stream_info_impl(TT *tt, std::array<stream_info_t, TT::numins>& streams) {
124 if constexpr (TT::numins > i) {
125 if (std::get<i>(tt->input_reducers)) {
126 streams[i].goal = tt->static_stream_goal[i];
127 streams[i].size = 0;
128 parsec_atomic_lock_init(&streams[i].reduce_copies_lock);
129 streams[i].reduce_copies.clear();
130 streams[i].reduce_count.store(0, std::memory_order_relaxed);
131 }
132 /* recursion */
133 if constexpr((i + 1) < TT::numins) {
134 init_stream_info_impl<i+1>(tt, streams);
135 }
136 }
137 }
